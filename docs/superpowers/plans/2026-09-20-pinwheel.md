**Interfaces:**
- Consumes: nothing.
- Produces:
  ```gdscript
  static func band(difficulty: int) -> Dictionary
      # {"cols": int, "rows": int, "sizes": Array[int], "min_turns": int, "max_stack": int}
  static func generate(rng: RandomNumberGenerator, difficulty: int) -> Dictionary
      # {
      #   "cols": int, "rows": int,
      #   "pins":   PackedInt32Array,   # per piece, the pin's cell index
      #   "shapes": Array,              # per piece: Array of orientations,
      #                                 #   each an Array[Vector2i] of absolute board cells,
      #                                 #   in clockwise order starting at the grown (solution) one
      #   "answer": PackedInt32Array,   # per piece, the solving orientation index (always 0)
      #   "start":  PackedInt32Array,   # per piece, the opening orientation index
      #   "cloth":  PackedInt32Array,   # per piece, its Pal.CLOTH index
      #   "unique": bool, "attempts": int, "turns": int,
      # }
  static func rotate(cells: Array, pin: Vector2i, quarters: int) -> Array
  static func orientations(cells: Array, pin: Vector2i, cols: int, rows: int) -> Array
  static func count_tilings(shapes: Array, cols: int, rows: int, cap: int) -> int
  ```

**Design notes the implementer needs:**

`shapes[p]` holds the piece's **distinct in-frame orientations, in clockwise order, with the solution first**. That is why `answer` is always 0 — it is kept in the dictionary anyway so the state class never has to know that, and so a future generator that grows a scramble rather than an answer does not break the consumer.

"In clockwise order starting at the solution" means: take `quarters = 0, 1, 2, 3` about the pin; keep the ones wholly inside the frame; drop exact duplicates (a 1×1 has one, a 2×2 square about its own corner has four distinct, a plus about its centre has one). A tap advances the index by one, modulo the count — that is the whole of rule 4, and it is why the list must stay in clockwise order.

```gdscript
extends RefCounted

## The generator's own suite: the geometry it rests on, the proof it makes,
## and the promise that a seed is a board.

const Gen = preload("res://puzzles/pinwheel_gen.gd")

static func run(t) -> void:
	_test_rotate_is_clockwise_about_the_pin(t)
	_test_orientations_drop_duplicates_and_escapees(t)
	_test_count_tilings_counts(t)
	_test_every_band_builds(t)
	_test_repeatable(t)
	_test_the_answer_tiles_the_frame(t)
	_test_the_start_is_reachable_and_not_the_answer(t)
	_test_touching_pieces_never_share_a_cloth(t)

static func _cells(raw: Array) -> Array:
	var out: Array = []
	for p in raw:
		out.append(Vector2i(p[0], p[1]))
	return out

static func _key(cells: Array) -> String:
	var ids: Array = []
	for c: Vector2i in cells:
		ids.append("%d,%d" % [c.x, c.y])
	ids.sort()
	return ";".join(ids)

static func _test_rotate_is_clockwise_about_the_pin(t) -> void:
	# A horizontal domino pinned at its left cell swings down, not up:
	# clockwise on a y-down screen sends (+1, 0) to (0, +1).
	var cells := _cells([[2, 3], [3, 3]])
	var turned: Array = Gen.rotate(cells, Vector2i(2, 3), 1)
	t.eq(_key(turned), _key(_cells([[2, 3], [2, 4]])), "one quarter turn is clockwise")
	t.eq(_key(Gen.rotate(cells, Vector2i(2, 3), 4)), _key(cells), "four quarters come home")

static func _test_orientations_drop_duplicates_and_escapees(t) -> void:
	# A lone cell has exactly one orientation however much it is turned.
	t.eq(Gen.orientations(_cells([[0, 0]]), Vector2i(0, 0), 5, 5).size(), 1,
		"a single cell has one orientation")
	# A domino pinned in the top-left corner can only lie right or hang down:
	# the other two quarters leave the frame.
	var corner: Array = Gen.orientations(_cells([[0, 0], [1, 0]]), Vector2i(0, 0), 5, 5)
	t.eq(corner.size(), 2, "a corner domino keeps two of its four quarters")
	# The same domino in the middle keeps all four.
	t.eq(Gen.orientations(_cells([[2, 2], [3, 2]]), Vector2i(2, 2), 5, 5).size(), 4,
		"a domino with room keeps four")

static func _test_count_tilings_counts(t) -> void:
	# Two dominoes in a 2x2 frame, each pinned in its own corner. There is
	# exactly one way to lay them without overlapping: both horizontal.
	var a: Array = Gen.orientations(_cells([[0, 0], [1, 0]]), Vector2i(0, 0), 2, 2)
	var b: Array = Gen.orientations(_cells([[0, 1], [1, 1]]), Vector2i(1, 1), 2, 2)
	t.eq(Gen.count_tilings([a, b], 2, 2, 5), 1, "a 2x2 of two pinned dominoes tiles one way")

static func _test_every_band_builds(t) -> void:
	for d in 3:
		var b: Dictionary = Gen.band(d)
		var missed := 0
		var unproved := 0
		for s in 40:
			var rng := RandomNumberGenerator.new()
			rng.seed = 7000 + s
			var g: Dictionary = Gen.generate(rng, d)
			if int(g.cols) == 0:
				missed += 1
				continue
			if not bool(g.unique):
				unproved += 1
			t.eq((g.shapes as Array).size(), (b.sizes as Array).size(),
				"band %d seed %d built every piece" % [d, s])
			t.check(int(g.turns) >= int(b.min_turns),
				"band %d seed %d is deep enough (%d taps)" % [d, s, int(g.turns)])
		t.eq(missed, 0, "band %d missed no seed" % d)
		t.eq(unproved, 0, "band %d proved every board unique" % d)

static func _test_repeatable(t) -> void:
	for d in 3:
		var one := RandomNumberGenerator.new()
		var two := RandomNumberGenerator.new()
		one.seed = 4242
		two.seed = 4242
		var a: Dictionary = Gen.generate(one, d)
		var b: Dictionary = Gen.generate(two, d)
		t.eq(str(a.pins), str(b.pins), "band %d: the same seed pins the same cells" % d)
		t.eq(str(a.start), str(b.start), "band %d: the same seed opens the same way" % d)
		t.eq(str(a.cloth), str(b.cloth), "band %d: the same seed colours the same" % d)

static func _test_the_answer_tiles_the_frame(t) -> void:
	for d in 3:
		for s in 12:
			var rng := RandomNumberGenerator.new()
			rng.seed = 900 + s
			var g: Dictionary = Gen.generate(rng, d)
			var cols := int(g.cols)
			var rows := int(g.rows)
			var seen := PackedInt32Array()
			seen.resize(cols * rows)
			for p in (g.shapes as Array).size():
				for c: Vector2i in ((g.shapes[p] as Array)[int(g.answer[p])] as Array):
					t.check(c.x >= 0 and c.x < cols and c.y >= 0 and c.y < rows,
						"band %d seed %d: the answer stays in the frame" % [d, s])
					seen[c.y * cols + c.x] += 1
			for i in seen.size():
				t.eq(seen[i], 1, "band %d seed %d cell %d is covered once" % [d, s, i])

static func _test_the_start_is_reachable_and_not_the_answer(t) -> void:
	# Every orientation is reachable because a tap steps round the cycle, so
	# the only thing to check is that the board does not open solved and that
	# a piece with somewhere to go is actually moved off its answer.
	for d in 3:
		for s in 12:
			var rng := RandomNumberGenerator.new()
			rng.seed = 300 + s
			var g: Dictionary = Gen.generate(rng, d)
			var moved := 0
			for p in (g.shapes as Array).size():
				var m: int = (g.shapes[p] as Array).size()
				t.check(int(g.start[p]) >= 0 and int(g.start[p]) < m,
					"band %d seed %d piece %d opens on a real orientation" % [d, s, p])
				if m > 1:
					t.check(int(g.start[p]) != int(g.answer[p]),
						"band %d seed %d piece %d does not open solved" % [d, s, p])
					moved += 1
			t.check(moved > 0, "band %d seed %d has something to turn" % [d, s])

static func _test_touching_pieces_never_share_a_cloth(t) -> void:
	# Two pieces that can ever meet must not repeat a colour, or the player
	# reads one shape where there are two.
	for d in 3:
		for s in 10:
			var rng := RandomNumberGenerator.new()
			rng.seed = 600 + s
			var g: Dictionary = Gen.generate(rng, d)
			var reach: Array = []
			for p in (g.shapes as Array).size():
				var near: Dictionary = {}
				for orient: Array in (g.shapes[p] as Array):
					for c: Vector2i in orient:
						near[c] = true
						for dd in [Vector2i(1, 0), Vector2i(-1, 0), Vector2i(0, 1), Vector2i(0, -1)]:
							near[c + dd] = true
				reach.append(near)
			for a in reach.size():
				for b in range(a + 1, reach.size()):
					if int(g.cloth[a]) != int(g.cloth[b]):
						continue
					var meets := false
					for c in (reach[b] as Dictionary):
						if (reach[a] as Dictionary).has(c):
							meets = true
							break
					t.check(not meets,
						"band %d seed %d: pieces %d and %d share a cloth and can meet" % [d, s, a, b])
```

- [ ] **Step 3: Write the generator**

Create `puzzles/pinwheel_gen.gd`. The algorithm is the spec's §5, and the reference implementation is `/tmp/pinwheel/gen.py` (Python) and `/tmp/pinwheel/engine.js` (JavaScript) — both are measured and both agree. Port, do not reinvent.

```gdscript
extends RefCounted

## Pinwheel's generator: grow an answer, pin it, prove it, scramble it.
##
## The board is a rectangle exactly tiled by polyominoes. Each piece is
## pinned through one of its own cells and can only rotate about that pin,
## which means **a piece has at most four placements in the whole frame** --
## it cannot translate at all. That is the opposite of Quilt, whose patches
## translate freely, and it is why the warning in `quilt_gen.gd`'s header
## ("a rotating patch would make almost every region tileable a dozen ways")
## does not apply here. Pinwheel is far more constrained than Quilt, and
## uniqueness comes almost free: the median board is proved on the first or
## second grow.
##
## Four stages, and each one can reject:
##   1. grow   -- partition the frame into polyominoes of the band's sizes
##   2. pin    -- choose each piece's pin to maximise its in-frame orientations
##   3. prove  -- exact cover, capped at two solutions; require exactly one
##   4. scramble -- the tidiest opening that still needs the band's taps
##
## There is no wall-clock budget and no `graded: false`. The loop is bounded
## by ATTEMPTS and the honest flag is `unique: false`, which is Quilt's
## answer rather than Sudoku's; a board that fails the proof is still
## playable, because the scramble is by construction reachable from a real
## tiling.

const DIRS := [Vector2i(1, 0), Vector2i(-1, 0), Vector2i(0, 1), Vector2i(0, -1)]

const BANDS := [
	{"cols": 5, "rows": 5, "sizes": [1, 2, 2, 3, 3, 3, 4, 4, 3], "min_turns": 8, "max_stack": 2},
	{"cols": 5, "rows": 7, "sizes": [1, 2, 2, 3, 3, 3, 4, 4, 4, 5, 4], "min_turns": 14, "max_stack": 3},
	{"cols": 6, "rows": 8, "sizes": [2, 2, 3, 3, 3, 4, 4, 4, 5, 5, 5, 4, 4], "min_turns": 20, "max_stack": 3},
]

const ATTEMPTS := 400     ## grows before the loop gives up
const SCRAMBLES := 220    ## openings tried per proved board, best kept
const CLOTHS := 8         ## Pal.CLOTH's size, mirrored here so the generator stays scene-free
```

Implement, in this order:

1. `band(difficulty)` — clamps into `BANDS`.
2. `rotate(cells, pin, quarters)` — `d = c - pin`, then `quarters` times `d = Vector2i(-d.y, d.x)` (clockwise on a y-down screen), then `pin + d`.
3. `orientations(cells, pin, cols, rows)` — the four rotations, dropping any with a cell outside the frame and any that repeats an earlier one's cell set. Returns `Array[Array[Vector2i]]` in quarters order, so index 0 is always the given `cells`.
4. `_grow(rng, b)` — for each size in a shuffled copy of `b.sizes`: find the free cells, score each by how many free orthogonal neighbours it has, seed at a random one of the minimum-scoring, then accrete at random from the piece's free frontier. Return `[]` if a piece wedges or a cell is left over. **Seeding into the tightest corner is load-bearing** — without it most grows wedge.
5. `_pin(rng, cells, cols, rows)` — the candidate pin cells that maximise `orientations(...).size()`, one chosen at random.
6. `count_tilings(shapes, cols, rows, cap)` — exact cover. Precompute, per cell, the list of `(piece, orientation)` that covers it. Recurse on the **first uncovered cell in scan order**: it must be covered by something, so try each `(piece, orientation)` that reaches it whose piece is unused and whose cells are all clear. Stop at `cap`.
7. `_scramble(rng, shapes, cols, rows, b)` — `SCRAMBLES` tries; each picks a random non-zero index for every piece with more than one orientation, sums `(0 - i) mod m` for the tap count, rejects below `b.min_turns`, builds a coverage count, rejects above `b.max_stack` deep, and scores `stained_cells * 10 + deepest`. Keep the lowest score. Return `{}` if none passed.
8. `_colour(shapes, cols, rows)` — build the "could ever meet" graph (piece A's every orientation's cells, dilated by one orthogonal step, intersected with piece B's undilated union), then greedy-colour in descending degree over `CLOTHS`. **If greedy runs out, fall back to `p % CLOTHS` for the remainder rather than looping** — a colour clash is a blemish, a generator that never returns is a crash.
9. `generate(rng, difficulty)` — the loop: grow, pin, prove, scramble; return on the first that passes all four. Keep the first proved-unique board seen as a fallback, and if `ATTEMPTS` runs out return that (or, failing everything, `{"cols": 0, ...}` with `unique: false`).

```gdscript
extends RefCounted

## Pinwheel's rules, scene-free.

const State = preload("res://puzzles/pinwheel_state.gd")

static func run(t) -> void:
	_test_a_tap_steps_round_the_cycle(t)
	_test_a_pinned_fast_piece_refuses_and_keeps_no_history(t)
	_test_the_stain_is_derived(t)
	_test_solved_only_when_every_cell_is_covered_once(t)
	_test_one_tap_one_undo(t)
	_test_hint_walks_a_piece_home_in_one_undo(t)
	_test_undo_does_not_refund_a_hint(t)
	_test_reset_clears_history(t)
	_test_the_answer_solves_every_band(t)

static func _state(seed_value: int, difficulty: int) -> State:
	var st := State.new()
	var rng := RandomNumberGenerator.new()
	rng.seed = seed_value
	st.setup(rng, difficulty)
	return st

static func _movable(st) -> int:
	for p in (st.shapes as Array).size():
		if not st.fixed(p):
			return p
	return -1

static func _test_a_tap_steps_round_the_cycle(t) -> void:
	var st := _state(11, 1)
	var p := _movable(st)
	t.check(p >= 0, "a board has something to turn")
	var m: int = (st.shapes[p] as Array).size()
	var was: int = st.turned[p]
	st.turn(p)
	t.eq(int(st.turned[p]), (was + 1) % m, "a tap advances one orientation")
	for i in m - 1:
		st.turn(p)
	t.eq(int(st.turned[p]), was, "m taps come home")

static func _test_a_pinned_fast_piece_refuses_and_keeps_no_history(t) -> void:
	var st := _state(11, 1)
	var fixed := -1
	for p in (st.shapes as Array).size():
		if st.fixed(p):
			fixed = p
			break
	if fixed < 0:
		return   # bands 0 and 1 carry a 1x1; band 2 may not. Nothing to assert.
	var before: int = (st.history as Array).size()
	t.eq(st.turn(fixed), false, "a pinned-fast piece refuses the tap")
	t.eq((st.history as Array).size(), before, "a refusal keeps no history")

static func _test_the_stain_is_derived(t) -> void:
	var st := _state(23, 1)
	var p := _movable(st)
	var before: Array = []
	for i in st.cols * st.rows:
		before.append(int(st.cover[i]))
	var m: int = (st.shapes[p] as Array).size()
	for i in m:
		st.turn(p)
	for i in st.cols * st.rows:
		t.eq(int(st.cover[i]), int(before[i]),
			"turning a piece all the way round leaves the stain where it was")

static func _test_solved_only_when_every_cell_is_covered_once(t) -> void:
	var st := _state(23, 1)
	t.eq(st.is_solved(), false, "a fresh board is not solved")
	for p in (st.shapes as Array).size():
		while int(st.turned[p]) != int(st.answer[p]):
			st.turn(p)
	t.eq(st.is_solved(), true, "the answer solves it")
	for i in st.cols * st.rows:
		t.eq(int(st.cover[i]), 1, "every cell is covered exactly once")

static func _test_one_tap_one_undo(t) -> void:
	var st := _state(31, 1)
	var p := _movable(st)
	var was: int = st.turned[p]
	st.turn(p)
	var u: Dictionary = st.undo()
	t.eq(int(u.get("piece", -1)), p, "undo names the piece")
	t.eq(int(st.turned[p]), was, "undo puts it back")
	t.eq(st.undo().is_empty(), true, "there is nothing left to undo")

static func _test_hint_walks_a_piece_home_in_one_undo(t) -> void:
	var st := _state(31, 1)
	var before: int = (st.history as Array).size()
	var h: Dictionary = st.hint()
	t.check(not h.is_empty(), "a hint has something to do on a fresh board")
	var p := int(h["piece"])
	t.eq(int(st.turned[p]), int(st.answer[p]), "a hint takes the piece all the way home")
	t.eq((st.history as Array).size(), before + 1, "however many quarters, a hint is one entry")
```

Create `puzzles/pinwheel_state.gd`, `extends RefCounted`, with the surface quoted in the Interfaces block. The rules that matter:

- `recompute()` rebuilds `cover` from scratch: zero it, then for every piece add one to each cell of `cells_of(p)`. `cover` is **derived and never stored** — that is what makes undo keep no book for the stain.
- `turn(p)` returns `false` and pushes nothing when `fixed(p)`; otherwise advances `turned[p]` by one modulo the orientation count, calls `recompute()`, pushes `{"piece": p, "from": was, "to": turned[p]}`, and returns `true`.
- `hint()` picks, among the pieces not already on their answer and not fixed, **the one furthest from home** (the most taps to come back) — the hint should be worth spending. Walk it to `answer[p]` in one go, push **one** history entry carrying `from` and `to`, `hints_used += 1`, `recompute()`. Return `{}` when nothing is wrong. Offer a hint even when `ok == false`, the way Quilt does: the stored answer is a tiling whatever the proof said.
- `undo()` pops and writes `turned[m.piece] = m.from`, `recompute()`, and returns the **inverted** move (`{"piece": p, "from": m.to, "to": m.from}`) so the board can animate it. It does **not** refund `hints_used`.
- `reset()` writes `start` back over `turned`, `recompute()`, **clears `history`**, and returns one `{"piece","from","to"}` per piece that actually moved.
- `share_glyphs()` — one line a row; a bare cell `"⬛"`, a covered cell one of `["🟨","🟦","🟥","🟩","🟪","🟧","🟫","🟪"]` indexed by the covering piece's `cloth`, a stained cell `"⬜"`.

```gdscript
## The angle a thing has swung through, `quarters` quarter turns clockwise,
## `elapsed` seconds after the moment it was set off. Reads `back_out`, so it
## passes the quarter and settles back onto it -- a pinwheel that stopped dead
## on the mark reads as a snap rather than a spin. This is the repo's first
## rotation recipe: every turn before it was either an idle (the sun's
## revolution, Untangle's knots) or Hidden Word's flip, which is a scale on
## one axis and not a rotation at all.
static func turn_angle(elapsed: float, quarters: int, time := TURN_TIME) -> float:
	var full := float(quarters) * PI * 0.5
	if reduce or elapsed >= time:
		return full
	if elapsed <= 0.0:
		return 0.0
	return full * back_out(elapsed / time)
```

1. **Layout.** `_cell()` is `minf((size.x - 2*INSET) / state.cols, (size.y - 2*INSET) / state.rows)`; `_origin()` centres the `cols x rows` block in the rect. **Do not pack a cell as `row * cols + column` anywhere a column could be out of range** — Quilt shipped that and a hold one cell off the left edge wrapped onto the far right, 2,386 of them coming back legal across 120 boards. 2. **Input.** `_gui_input` on a press: convert to a cell, bounds-check, then `state.piece_at_pin(c, r)`.

4. **`_draw`.** Build any null mesh, `draw_mesh` it, append it to a local `shown` array, and assign `_shown = shown` at the end. Guard every builder with "an empty builder cannot be asked for a mesh" (`b.mesh() if not b.verts.is_empty() else null`).

7. **The refusal.** `Motion.shiver_offset` on the piece's position and a `Pal.BAD` halo stroked round its silhouette at `Motion.flash_level`. **Never a wash into the cloth.**
8. **Host actions.** `can_undo()`, `undo()`, `hints_left()`, `hint()`, `reset_board()` — each takes a `cover` snapshot first and routes through `_settle`; `hint()` also `fx.ring` and `fx.sparkle` at the piece's pin, and `hints_used += 1`. `undo()` and `hint()` call `check_solved()` directly and **not** `note_move()` — they are not moves.
